#----app.java---------
public class app {
    public static void main(String[] args) {
        System.out.println("Hello from app.java inside Docker!");
    }
}

#-------------Dockerfile-----------
FROM eclipse-temurin:21-jdk
WORKDIR /app
COPY app.java .
RUN javac app.java
CMD ["java", "app"]

#-------------buildcommands-----------

# Build the Docker image
docker build -t java-app .

# Run the container
docker run --rm java-app
