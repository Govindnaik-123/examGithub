#--------------parameterizedConstructor-----------
class Student {
    String name;
    int rollNo;

    Student(String name, int rollNo) {
        this.name = name;
        this.rollNo = rollNo;
    }

    void display() {
        System.out.println("Name: " + name + ", Roll No: " + rollNo);
    }
}

public class app {
    public static void main(String[] args) {
        Student s1 = new Student("Govind", 16);
        s1.display();
    }
}
--------
docker build -t java-ctor-app .
docker run --rm java-ctor-app
--------

#--------------------inheritance--------------

class Vehicle {
    void start() {
        System.out.println("Vehicle started.");
    }
}

class Bike extends Vehicle {
    void ride() {
        System.out.println("Bike is riding.");
    }
}

public class app {
    public static void main(String[] args) {
        Bike b = new Bike();
        b.start();
        b.ride();
    }
}
----------
docker build -t java-inheritance-app .
docker run --rm java-inheritance-app
-----------


#---------Palindrome---------------------------
public class app {
    public static void main(String[] args) {
        int n = 121;
        int temp = n;
        int rev = 0;

        while (n > 0) {
            int r = n % 10;
            rev = rev * 10 + r;
            n = n / 10;
        }

        if (temp == rev) {
            System.out.println(temp + " is a palindrome");
        } else {
            System.out.println(temp + " is not a palindrome");
        }
    }
}

-------------
docker build -t java-palindrome .
docker run --rm java-palindrome
------------
